### 3.2 `OneBotChannel`（`src/platforms/onebot.py`）

```python
async def get_applicant_profile(self, request, *, caller="join_review") -> dict:
    uid = str(request.get("member_openid") or request.get("user_id") or "")
    profile = ApplicantProfile(platform_id=self.platform_id, kind="onebot", user_id=uid,
                               source="onebot_stranger", degraded=True)
    if not uid.isdigit():
        profile.note = "缺少 QQ 号，无法获取资料"
        return profile.to_dict()
    try:
        info = await self._call("get_stranger_info", user_id=_int(uid), no_cache=False)
    except Exception as exc:
        profile.note = f"get_stranger_info 失败：{type(exc).__name__}"
        return profile.to_dict()
    payload = info if isinstance(info, dict) else {}
    profile.nickname = _text(payload.get("nickname") or payload.get("long_nick") or payload.get("remark"))
    profile.qq_level = _level(payload.get("qqLevel"))        # 宽进：int/str/{"level":n}/None
    profile.qid = _text(payload.get("qid"))
    profile.sex = _text(payload.get("sex"))
    profile.age = _opt_int(payload.get("age"))
    profile.reg_time = _opt_int(payload.get("reg_time") or payload.get("regTime"))
    if profile.reg_time:
        profile.account_age_days = max(0, (now_ts() - profile.reg_time) // 86400)
    profile.is_vip = bool(payload.get("is_vip"))
    profile.vip_level = _opt_int(payload.get("vip_level")) or 0
    profile.avatar_url = f"https://q1.qlogo.cn/g?b=qq&nk={uid}&s=640"
    profile.degraded = not (profile.nickname and profile.has_account_signals)
    profile.note = "" if not profile.degraded else "协议端未返回昵称或账号等级/注册时间"
    profile.fetched_at = now_ts()
    return profile.to_dict()
```

要点：
- `_level()` 必须容忍 NapCat 的 `qqLevel` 为数字/字符串/对象三种形态（`{"level": n}` 之类）。
- **只读、不写**，`dry_run` 不适用（`WRITE_ACTIONS` 不含 `get_stranger_info`）。
- 头像直链与 NapCat 内部一致（`q.qlogo.cn/headimg_dl` 同源 CDN）。

### 3.4 `src/profiles.py`（新增）

```python
class ApplicantProfileService:
    def __init__(self, *, store, channel_for, logger=None, clock=time.monotonic) -> None: ...

    async def get(self, group_id: str, request: dict) -> dict:
        """返回画像 dict；永不抛异常。"""
        settings = self.store.settings()
        if not settings.get("join_profile_enabled", True):
            return ApplicantProfile(degraded=True, note="画像采集已关闭").to_dict()

        platform_id = (self.store.group(group_id).platform_id if self.store.group(group_id) else "")
        uid = str(request.get("user_id") or request.get("member_openid") or "")
        cache_key = f"{platform_id}:{uid}"

        # 1) 缓存（只缓存成功画像）
        cached = self._cache_get(cache_key, ttl_days=int(settings.get("join_profile_cache_days", 7)))
        if cached:
            cached["source"] = "cache"
            return cached
        # 2) 熔断
        if self._breaker_open():
            return ApplicantProfile(degraded=True, note="画像服务熔断中").to_dict()
        # 3) 限频 + 并发
        await self._acquire(settings)
        try:
            channel = self.channel_for(platform_id)
            profile = await channel.get_applicant_profile(request)
        except Exception as exc:                      # 兜底：任何异常都降级
            profile = ApplicantProfile(degraded=True, note=f"画像采集异常：{type(exc).__name__}").to_dict()
        finally:
            self._release()
        self._record_result(profile)
        if not profile.get("degraded"):
            await self.store.put_profile(cache_key, profile)
        return profile
```

内部件：
- `_cache_get`：内存优先，回退 `store.profile_cache()`；TTL 用 `fetched_at` 判断。
- 令牌桶：容量 `join_profile_qpm`，按分钟补充；`asyncio.sleep` 等待而不是丢弃。
- 并发：`asyncio.Semaphore(join_profile_concurrency)`，懒初始化（避免事件循环问题）。
- 熔断：连续失败 `>= 5` 开 300s，之后半开重试 1 次（复用 `circuit_break_threshold` 语义，独立计数）。

### 4.2 `judge()` 新增步骤

在 `risk_tips == "top_tips"` 之后、信任邀请人之前插入：

```python
profile = request.get("profile") or {}
gate = self._profile_gate(profile, settings)
if gate is not None:
    return gate
```

```python
def _profile_gate(self, profile: dict, settings: dict) -> JoinDecision | None:
    if profile.get("degraded"):
        mode = str(settings.get("join_profile_missing") or "manual")
        if mode == "pass":
            return None
        if mode == "decline":
            return JoinDecision(op="decline", auto=True, confidence=1.0,
                                reason="无法获取申请人资料（按策略拒绝）", risk="其他",
                                source="rule", gate="profile_missing")
        return JoinDecision(op="approve", auto=False,
                            reason="资料缺失，转人工复核", source="manual", gate="profile_missing")

    action = str(settings.get("join_gate_action") or "decline")
    limit_days = int(settings.get("join_min_account_days", 0) or 0)
    age = profile.get("account_age_days")
    if limit_days > 0 and isinstance(age, int) and age < limit_days:
        return self._gate_decision(action, f"账号注册仅 {age} 天（< {limit_days} 天）", "account_age")
    limit_level = int(settings.get("join_min_qq_level", 0) or 0)
    level = profile.get("qq_level")
    if limit_level > 0 and isinstance(level, int) and level < limit_level:
        return self._gate_decision(action, f"QQ 等级 {level}（< {limit_level}）", "qq_level")
    if settings.get("join_require_qid") and not str(profile.get("qid") or "").strip():
        return self._gate_decision(action, "缺少 QID", "qid")
    return None
```

`_gate_decision(action, reason, gate)`：`decline` → 自动拒绝 + 按 `join_decline_blacklist` 决定拉黑；`manual` → `auto=False`；`pass` → `None`（继续走后续链路）。
**关键**：字段为 `None` 时**不命中**规则（缺失由第 5 步统一处理），避免把「协议端不支持」当成「可疑」。

### 4.3 `JoinDecision`

新增 `gate: str = ""`，`to_dict()` 带上；`_persist` 写入 `join_requests.gate`。

```python
async def _call_judge(self, system_prompt: str, user_prompt: str, image_urls: list[str]) -> str:
```

### 5.2 头像复核（单向收紧）

```python
mode = str(settings.get("join_avatar_review") or "off")

below = float(settings.get("join_avatar_only_below", 0.95) or 0.95)
if mode != "off" and avatar and (mode == "always" or confidence < below):
    raw2 = await self._call_judge(AVATAR_SYSTEM_PROMPT, AVATAR_USER_TEMPLATE.format(...), [avatar])
    avatar_verdict = self._parse_avatar(raw2)        # {"risk": bool, "confidence": float, "reason": str}
```

### 11.4 真机验证（已完成）

环境：远端 txyun 49.232.95.208，NapCat **v4.18.28** + QQ 3.2.32 + AstrBot 4.28.0（OneBot11 反向 WS）。
方法：SSH 隧道把 NapCat WebUI（127.0.0.1:6099）映射到本机 → 登录接口（sha256(webui_token + 后缀)）取凭证 → 建调试适配器 → 执行 OneBot 只读动作。
